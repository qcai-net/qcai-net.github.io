import sys
import argparse
import json
import time
import ctypes
import threading
import os

# --- Watchdog Timer ---
def start_watchdog(timeout_sec=30):
    def timeout_handler():
        sys.stderr.write(f"\n[FATAL ERROR] WeChat Bridge timed out. UI Automation hung.\n")
        os._exit(1)
    timer = threading.Timer(timeout_sec, timeout_handler)
    timer.daemon = True
    timer.start()

def log(msg):
    sys.stdout.write(f"DEBUG: {msg}\n")
    sys.stdout.flush()

def bring_wechat_to_foreground():
    user32 = ctypes.windll.user32
    hwnd = user32.FindWindowW("Qt51514QWindowIcon", None)
    if hwnd:
        user32.ShowWindow(hwnd, 9)   # SW_RESTORE
        user32.SetForegroundWindow(hwnd)
        time.sleep(1.0)
        return True
    return False

def patch_sessionbox_search():
    import wxauto4.ui.sessionbox as sb
    def patched_search(self, keywords, force=False, force_wait=0.5):
        from wxauto4.utils.win32 import SetClipboardText
        self.searchbox.Click()
        time.sleep(0.3)
        self.searchbox.SendKeys('{Ctrl}a', interval=0.1)
        self.searchbox.SendKeys('{Delete}', interval=0.1)
        SetClipboardText(keywords)
        self.searchbox.SendKeys('{Ctrl}v', interval=0.1)
        time.sleep(2.0)
        try:
            search_result_list = self.search_content.ListControl()
            children = search_result_list.GetChildren()
            candidates = []
            for child in children:
                name = str(child.Name).strip()
                if any(w in name for w in ["Internet", "Search", "朋友圈", "小程序"]): continue
                score = sum(1 for char in keywords if char in name)
                if score > 0: candidates.append((score, child))
            candidates.sort(key=lambda x: x[0], reverse=True)
            if candidates:
                best_match = candidates[0][1]
                log(f"Clicking: {best_match.Name}")
                try:
                    best_match.Click(simulateMove=False)
                except:
                    import ctypes
                    user32 = ctypes.windll.user32
                    rect = best_match.BoundingRectangle
                    user32.SetCursorPos((rect.left + rect.right) // 2, (rect.top + rect.bottom) // 2)
                    user32.mouse_event(0x0002, 0, 0, 0, 0)
                    user32.mouse_event(0x0004, 0, 0, 0, 0)
                time.sleep(1.5)
                class MockElement:
                    def Click(self, *args, **kwargs): pass
                class MockResult:
                    def __init__(self): self.element = MockElement()
                    def Click(self): pass
                return [MockResult()]
            return []
        except: return []
    sb.SessionBox.search = patched_search

def parse_message(m):
    try:
        sender = str(getattr(m, 'sender', m[0] if isinstance(m, tuple) else "Unknown"))
        content = str(getattr(m, 'content', m[1] if isinstance(m, tuple) else ""))
        return {"sender": sender, "content": content}
    except: return {"sender": "error", "content": "Parse error"}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--action', choices=['send', 'sync'], required=True)
    parser.add_argument('--who', required=True)
    parser.add_argument('--msg', default='')
    args = parser.parse_args()

    start_watchdog(30)
    if not bring_wechat_to_foreground():
        print(json.dumps({"status": "error", "message": "WeChat Desktop not found."}))
        sys.exit(1)

    patch_sessionbox_search()
    from wxauto4 import WeChat
    wx = WeChat()

    # Delay Patch
    original_chat_with = wx.ChatWith
    wx.ChatWith = lambda *a, **kw: (original_chat_with(*a, **kw), time.sleep(2.0))[0]

    if args.action == 'sync':
        try:
            wx.ChatWith(args.who, exact=False)
            msgs = wx.GetAllMessage()
            output = [parse_message(m) for m in msgs]
            print(json.dumps({"status": "success", "messages": output}, ensure_ascii=False))
        except Exception as e: print(json.dumps({"status": "error", "message": str(e)}))

    elif args.action == 'send':
        try:
            wx.ChatWith(args.who, exact=False)
            from wxauto4.utils.win32 import SetClipboardText
            SetClipboardText(args.msg)
            time.sleep(1.0)
            chat_api = wx._api._chat_api
            chat_api._activate_editbox()
            chat_api.editbox.SendKeys('{Ctrl}v', waitTime=0.5)
            chat_api.editbox.SendKeys('{Enter}', waitTime=0.5)
            print(json.dumps({"status": "success", "detail": f"Sent to '{args.who}'"}))
        except Exception as e: print(json.dumps({"status": "error", "message": str(e)}))

if __name__ == "__main__":
    main()

import http.server
import socketserver
import json
import subprocess
import os

PORT = 18790

class PortalHandler(http.server.SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        super().end_headers()

    def do_OPTIONS(self):
        self.send_response(200)
        self.end_headers()

    def do_POST(self):
        if self.path == '/api/exec':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            data = json.loads(post_data.decode('utf-8'))
            cmd = data.get('cmd')
            try:
                skills_dir = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", "skills", "wechat"))
                result = subprocess.run(cmd, shell=True, capture_output=True, text=True, cwd=skills_dir)
                try: response_data = json.loads(result.stdout)
                except: response_data = {"status": "success", "output": result.stdout}
                self.send_response(200)
                self.send_header('Content-type', 'application/json')
                self.end_headers()
                self.wfile.write(json.dumps(response_data).encode('utf-8'))
            except Exception as e:
                self.send_response(500)
                self.end_headers()
                self.wfile.write(json.dumps({"status": "error", "message": str(e)}).encode('utf-8'))

if __name__ == "__main__":
    os.chdir(os.path.dirname(os.path.abspath(__file__)))
    with socketserver.TCPServer(("", PORT), PortalHandler) as httpd:
        print(f"Server on http://127.0.0.1:{PORT}")
        httpd.serve_forever()
