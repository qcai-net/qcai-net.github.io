Invoke this skill automatically when the user asks about:
- Sending a message to someone on WeChat
- Reading, syncing, or fetching the latest messages from a WeChat group or contact
- Replying to a WeChat conversation

**Trigger phrases:** "Send a WeChat message", "Sync WeChat", "Read WeChat", "message on WeChat"

import sys
import argparse
import json
import time
import ctypes
import threading
import os

# --- Watchdog Timer ---
def start_watchdog(timeout_sec=30):
    def timeout_handler():
        sys.stderr.write(f"\n[FATAL ERROR] WeChat Bridge timed out after {timeout_sec}s. UI Automation hung.\n")
        sys.stderr.flush()
        os._exit(1)
    timer = threading.Timer(timeout_sec, timeout_handler)
    timer.daemon = True
    timer.start()
    return timer

def log(msg):
    sys.stdout.write(f"DEBUG: {msg}\n")
    sys.stdout.flush()

def bring_wechat_to_foreground():
    log("Checking for WeChat window...")
    user32 = ctypes.windll.user32
    hwnd = user32.FindWindowW("Qt51514QWindowIcon", None)
    if hwnd:
        log(f"Found WeChat window (HWND: {hwnd}). Bringing to foreground...")
        user32.ShowWindow(hwnd, 9)   # SW_RESTORE
        user32.SetForegroundWindow(hwnd)
        time.sleep(1.0)
        return True
    return False

def patch_sessionbox_search():
    import wxauto4.ui.sessionbox as sb
    def patched_search(self, keywords, force=False, force_wait=0.5):
        from wxauto4.utils.win32 import SetClipboardText
        self.searchbox.Click()
        time.sleep(0.3)
        self.searchbox.SendKeys('{Ctrl}a', interval=0.1)
        self.searchbox.SendKeys('{Delete}', interval=0.1)
        SetClipboardText(keywords)
        self.searchbox.SendKeys('{Ctrl}v', interval=0.1)
        wait_time = force_wait if force else 2.0
        time.sleep(wait_time)
        try:
            search_result_list = self.search_content.ListControl()
            children = search_result_list.GetChildren()
            candidates = []
            for child in children:
                name = str(child.Name).strip()
                cls = str(child.ClassName).strip()
                blacklisted = ["Internet", "Search", "results", "搜一搜", "搜索", "相关结果", "公众号"]
                if cls == "mmui::XTextView" or any(w in name for w in blacklisted):
                    continue
                score = sum(1 for char in keywords if char in name)
                if score > 0: candidates.append((score, child))
            candidates.sort(key=lambda x: x[0], reverse=True)
            if candidates:
                best_match = candidates[0][1]
                try:
                    best_match.Click(simulateMove=False)
                except:
                    import ctypes
                    user32 = ctypes.windll.user32
                    rect = best_match.BoundingRectangle
                    x, y = (rect.left + rect.right) // 2, (rect.top + rect.bottom) // 2
                    user32.SetCursorPos(x, y)
                    user32.mouse_event(0x0002, 0, 0, 0, 0) 
                    user32.mouse_event(0x0004, 0, 0, 0, 0)
                time.sleep(1.5)
                class MockResult:
                    def __init__(self): self.element = type('',(object,),{'Click':lambda self:None})()
                    def Click(self): pass
                return [MockResult()]
            return []
        except: return []
    sb.SessionBox.search = patched_search

def parse_message(m):
    try:
        if hasattr(m, 'content'):
            return {"sender": str(getattr(m, 'sender', "Unknown")), "content": str(getattr(m, 'content', ""))}
        return {"sender": str(m[0]), "content": str(m[1])}
    except: return {"sender": "error", "content": "Parse error"}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--action', choices=['send', 'sync'], required=True)
    parser.add_argument('--who', required=True)
    parser.add_argument('--msg', default='')
    args = parser.parse_args()
    start_watchdog(30)
    if not bring_wechat_to_foreground():
        print(json.dumps({"status": "error", "message": "WeChat Desktop is not visible"}))
        sys.exit(1)
    patch_sessionbox_search()
    from wxauto4 import WeChat
    try:
        wx = WeChat()
    except Exception as e:
        print(json.dumps({"status": "error", "message": str(e)}))
        sys.exit(1)
    original_chat_with = wx.ChatWith
    def delayed_chat_with(*a, **kw):
        res = original_chat_with(*a, **kw)
        time.sleep(2.0)
        return res
    wx.ChatWith = delayed_chat_with
    if args.action == 'sync':
        try:
            wx.ChatWith(args.who, exact=False)
            time.sleep(2.0)
            msgs = wx.GetAllMessage()
            output = [parse_message(m) for m in msgs]
            print(json.dumps({"status": "success", "group": args.who, "messages": output}, ensure_ascii=False))
        except Exception as e: print(json.dumps({"status": "error", "message": str(e)}))
    elif args.action == 'send':
        try:
            wx.ChatWith(args.who, exact=False)
            user32 = ctypes.windll.user32
            hwnd = user32.FindWindowW("Qt51514QWindowIcon", None)
            if hwnd: user32.SetForegroundWindow(hwnd); time.sleep(0.5)
            try:
                wx.SendMsg(args.msg)
                print(json.dumps({"status": "success", "detail": f"Sent to '{args.who}'"}))
                return
            except: pass
            from wxauto4.utils.win32 import SetClipboardText
            SetClipboardText(args.msg)
            time.sleep(0.8)
            try:
                chat_api = wx._api._chat_api
                chat_api._activate_editbox()
                rect = chat_api.editbox.BoundingRectangle
                user32.SetCursorPos((rect.left+rect.right)//2, (rect.top+rect.bottom)//2)
                user32.mouse_event(0x0002, 0, 0, 0, 0); user32.mouse_event(0x0004, 0, 0, 0, 0)
                time.sleep(0.5)
            except: pass
            user32.keybd_event(0x11, 0, 0, 0); user32.keybd_event(0x56, 0, 0, 0)
            time.sleep(0.3)
            user32.keybd_event(0x56, 0, 2, 0); user32.keybd_event(0x11, 0, 2, 0)
            time.sleep(0.5)
            user32.keybd_event(0x0D, 0, 0, 0); time.sleep(0.1); user32.keybd_event(0x0D, 0, 2, 0)
            print(json.dumps({"status": "success", "detail": f"Sent to '{args.who}'"}))
        except Exception as e: print(json.dumps({"status": "error", "message": str(e)}))

if __name__ == "__main__":
    main()

import http.server
import socketserver
import json
import subprocess
import os

PORT = 18790

class PortalHandler(http.server.SimpleHTTPRequestHandler):
    def end_headers(self):
        # Enable CORS for local testing
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        super().end_headers()

    def do_OPTIONS(self):
        self.send_response(200)
        self.end_headers()

    def do_POST(self):
        if self.path == '/api/exec':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            data = json.loads(post_data.decode('utf-8'))
            
            cmd = data.get('cmd')
            if not cmd:
                self.send_response(400)
                self.end_headers()
                self.wfile.write(b'{"status": "error", "message": "No command provided"}')
                return

            try:
                skills_dir = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", "skills", "wechat"))

                # Inject -u for unbuffered Python output (prevents subprocess output hangs)
                import re
                unbuffered_cmd = re.sub(r'(python(?:3|[\.\d]*|\.exe)?)\s', r'\1 -u ', cmd, count=1, flags=re.IGNORECASE)

                result = subprocess.run(
                    unbuffered_cmd,
                    shell=True,
                    capture_output=True,
                    text=True,
                    cwd=skills_dir,
                    timeout=35
                )

                # Try to parse stdout as JSON
                try:
                    response_data = json.loads(result.stdout)
                except ValueError:
                    response_data = {
                        "status": "error",
                        "output": result.stdout or "(no stdout)",
                        "debug": result.stderr or ""
                    }

                self.send_response(200)
                self.send_header('Content-type', 'application/json')
                self.end_headers()
                self.wfile.write(json.dumps(response_data).encode('utf-8'))
            except subprocess.TimeoutExpired:
                self.send_response(200)
                self.send_header('Content-type', 'application/json')
                self.end_headers()
                self.wfile.write(json.dumps({"status": "error", "message": "Bridge timed out (35s). Is WeChat visible and not minimized?"}).encode('utf-8'))
            except Exception as e:
                self.send_response(500)
                self.send_header('Content-type', 'application/json')
                self.end_headers()
                self.wfile.write(json.dumps({"status": "error", "message": str(e)}).encode('utf-8'))
        else:
            self.send_response(404)
            self.end_headers()

if __name__ == "__main__":
    # Ensure we are in the portal directory to serve its files
    os.chdir(os.path.dirname(os.path.abspath(__file__)))
    with socketserver.TCPServer(("", PORT), PortalHandler) as httpd:
        print(f"ClawPortal Server running at http://127.0.0.1:{PORT}")
        httpd.serve_forever()

        const statusEl = document.getElementById('loading');
        const chatArea = document.getElementById('chat-area');

        function appendMessage(sender, content, type) {
            const div = document.createElement('div');
            div.className = `message ${type}`;
            div.innerHTML = `<span class="sender">${sender}</span>${content}`;
            chatArea.appendChild(div);
            chatArea.scrollTop = chatArea.scrollHeight;

        async function executeAction(action) {
            const who = document.getElementById('target').value;
            const msg = document.getElementById('message').value;

            if (!who) {
                alert("Enter a WeChat name first");
                return;
            }

            statusEl.style.display = 'flex';
            
            // Use absolute path for safety in the portal environment
            let cmd = `C:\\Users\\yonwang\\AppData\\Local\\Programs\\Python\\Python312\\python.exe wechat_bridge.py --action ${action} --who "${who}"`;
            if (action === 'send') cmd += ` --msg "${msg}"`;

            try {
                const resp = await fetch('/api/exec', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ cmd: cmd })
                });
                const data = await resp.json();
                
                if (action === 'send') {
                    appendMessage('Me', msg, 'sent');
                } else if (data.messages) {
                    chatArea.innerHTML = ''; // Clear for fresh sync
                    data.messages.forEach(m => {
                        const type = (m.sender === 'Me' || m.sender === 'Unknown') ? 'sent' : 'received';
                        appendMessage(m.sender, m.content, type);
                    });
                }
            } catch (e) {
                alert("Portal connection failed. Ensure dashboard.py is running.");
            } finally {
                statusEl.style.display = 'none';
            }
        }

        function executeSync() {
            executeAction('sync');
        }
