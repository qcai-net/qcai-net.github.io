import sys
import argparse
import json
import time
import ctypes
import threading
import os

# --- Watchdog Timer ---
def start_watchdog(timeout_sec=30):
    def timeout_handler():
        sys.stderr.write(f"\n[FATAL ERROR] WeChat Bridge timed out after {timeout_sec}s. UI Automation hung.\n")
        sys.stderr.flush()
        # Force terminate because UIAutomation often blocks signal handlers
        os._exit(1)
    
    timer = threading.Timer(timeout_sec, timeout_handler)
    timer.daemon = True
    timer.start()
    return timer

def log(msg):
    """Unbuffered logging to stdout."""
    sys.stdout.write(f"DEBUG: {msg}\n")
    sys.stdout.flush()

def bring_wechat_to_foreground():
    """Restore and focus the WeChat window before UI automation."""
    log("Checking for WeChat window...")
    user32 = ctypes.windll.user32
    hwnd = user32.FindWindowW("Qt51514QWindowIcon", None)
    if hwnd:
        log(f"Found WeChat window (HWND: {hwnd}). Bringing to foreground...")
        user32.ShowWindow(hwnd, 9)   # SW_RESTORE
        user32.SetForegroundWindow(hwnd)
        time.sleep(1.0)              # Let focus settle
        return True
    log("WeChat window NOT found.")
    return False

def patch_sessionbox_search():
    """
    Replace wxauto4's right-click-paste approach with direct SendKeys typing.
    This avoids the infinite hang caused by the right-click context menu
    never appearing when WeChat is not the active OS window.
    """
    import wxauto4.ui.sessionbox as sb
    log("Applying monkey-patch to SessionBox.search...")

    def patched_search(self, keywords, force=False, force_wait=0.5):
        from wxauto4.utils.win32 import SetClipboardText
        log(f"Searching for: {keywords}")
        # Click the search box to focus it
        self.searchbox.Click()
        time.sleep(0.3)
        # Clear any existing text, then use Ctrl+V to paste for reliable emoji support
        self.searchbox.SendKeys('{Ctrl}a', interval=0.1)
        self.searchbox.SendKeys('{Delete}', interval=0.1)
        SetClipboardText(keywords)
        self.searchbox.SendKeys('{Ctrl}v', interval=0.1)
        
        # Wait for search results window to populate
        wait_time = force_wait if force else 2.0
        log(f"Waiting {wait_time}s for search results...")
        time.sleep(wait_time)
        
        try:
            search_result_list = self.search_content.ListControl()
            children = search_result_list.GetChildren()
            log(f"Found {len(children)} raw search results.")
            
            candidates = []
            for child in children:
                name = str(child.Name).strip()
                cls = str(child.ClassName).strip()
                log(f"Evaluating: '{name}' (Type: {cls})")

                # 1. AGGRESSIVE SUBSTRING EXCLUSIONS
                blacklisted_words = [
                    "Internet", "Search", "results", "Results", "搜一搜", 
                    "搜索", "相关结果", "最常使用", "朋友圈", "小程序", "公众号", "表情"
                ]
                
                # Check for headers or web search labels
                if cls == "mmui::XTextView" or any(w in name for w in blacklisted_words):
                    log(f"  -> Rule: Skip (Blacklisted/Header): {name}")
                    continue
                
                # 2. SCORING (Overlap)
                score = 0
                for char in keywords:
                    if char in name:
                        score += 1
                
                log(f"  -> Candidate Item: '{name}' Score: {score}")
                if score > 0:
                    candidates.append((score, child))
            
            # Sort by score (highest first)
            candidates.sort(key=lambda x: x[0], reverse=True)
            
            if candidates:
                best_match = candidates[0][1]
                log(f"Explicitly clicking best match: {best_match.Name}")
                try:
                    best_match.Click(simulateMove=False)
                except Exception as e:
                    log(f"UIAutomation click failed, trying fallback: {str(e)}")
                    # Fallback to pure OS mouse click if UI tree prevents simulateMove
                    import ctypes
                    user32 = ctypes.windll.user32
                    rect = best_match.BoundingRectangle
                    x = (rect.left + rect.right) // 2
                    y = (rect.top + rect.bottom) // 2
                    user32.SetCursorPos(x, y)
                    user32.mouse_event(0x0002, 0, 0, 0, 0) # LEFTDOWN
                    user32.mouse_event(0x0004, 0, 0, 0, 0) # LEFTUP
                    
                time.sleep(1.5) # Wait for chat panel to render
                
                class MockElement:
                    def Click(self, *args, **kwargs): pass
                class MockResult:
                    def __init__(self):
                        self.element = MockElement()
                    def Click(self): pass
                return [MockResult()]
            else:
                return []
        except Exception as e:
            log(f"Error parsing search results: {str(e)}")
            return []

    sb.SessionBox.search = patched_search

def parse_message(m):
    """Safely extract sender/content from a wxauto4 Message object or tuple."""
    try:
        # If it's the newer object format
        if hasattr(m, 'content'):
            sender = str(getattr(m, 'sender', "Unknown"))
            content = str(getattr(m, 'content', ""))
            return {"sender": sender, "content": content}
        
        # If it's the older tuple format
        sender = str(m[0]) if len(m) > 0 else "Unknown"
        content = str(m[1]) if len(m) > 1 else ""
        return {"sender": sender, "content": content}
    except Exception as e:
        return {"sender": "error", "content": f"Parse error: {str(e)}"}

def main():
    parser = argparse.ArgumentParser(description="QCAI WeChat Bridge")
    parser.add_argument('--action', choices=['send', 'sync'], required=True)
    parser.add_argument('--who', required=True, help='Contact or group name')
    parser.add_argument('--msg', default='', help='Message text (for send action)')
    args = parser.parse_args()

    # Step 0: Start watchdog to prevent infinite hangs
    start_watchdog(30)

    # Step 1: Bring WeChat window to front
    if not bring_wechat_to_foreground():
        print(json.dumps({
            "status": "error",
            "message": "WeChat Desktop is not open or minimized to tray. Please make it visible."
        }))
        sys.exit(1)

    # Step 2: Patch search before importing WeChat
    patch_sessionbox_search()

    from wxauto4 import WeChat
    log("Initializing WeChat automation...")
    try:
        wx = WeChat()
    except Exception as e:
        print(json.dumps({"status": "error", "message": f"Initialization failed: {str(e)}"}))
        sys.exit(1)

    # Monkey-patch wx.ChatWith to universally guarantee WeChat finishes UI transitions!
    # This solves the 'typed in different recipient window' bug since the UI takes time to catch up.
    original_chat_with = wx.ChatWith
    def delayed_chat_with(*a, **kw):
        res = original_chat_with(*a, **kw)
        time.sleep(2.0) # Wait 2 seconds for the chat panel to physically appear
        return res
    wx.ChatWith = delayed_chat_with

    # Step 3: Execute Action
    if args.action == 'sync':
        try:
            log(f"Navigating to '{args.who}'...")
            wx.ChatWith(args.who, exact=False)
            log("Navigation finished. Waiting for messages...")
            time.sleep(2.0) # Allow chat pane to refresh

            # Verify we are on the correct chat
            try:
                current_chat_name = wx.GetSessionList()[0]
                log(f"Active chat is now: '{current_chat_name}'")
            except Exception as e:
                log(f"Failed to verify chat title: {str(e)}")
                current_chat_name = "Unknown"

            log("Fetching all messages from visible pane...")
            msgs = wx.GetAllMessage()
            log(f"Fetched {len(msgs)} records.")

            if not msgs:
                print(json.dumps({
                    "status": "warning",
                    "message": f"No messages found in '{args.who}'. Verify group name or visibility.",
                    "messages": []
                }))
                return

            output = [parse_message(m) for m in msgs]
            print(json.dumps({
                "status": "success",
                "group": args.who,
                "count": len(output),
                "messages": output
            }, ensure_ascii=False))

        except Exception as e:
            print(json.dumps({"status": "error", "message": str(e)}))

    elif args.action == 'send':
        if not args.msg:
            print(json.dumps({"status": "error", "message": "Missing --msg argument"}))
            return
        
        try:
            log(f"Sending to '{args.who}'...")
            
            # Switch to chat and natively type the message using wxauto4's SendMsg.
            # Our perfectly patched ChatWith guarantees it clicks the right element and sleeps 2.0s 
            log(f"Sending to '{args.who}' via explicit Enter strike...")
            
            # Switch to chat accurately using our improved search patch
            # This works flawlessly and leaves the chat window open and input focused!
            wx.ChatWith(args.who, exact=False)
            
            # Use raw Windows API to copy text to clipboard securely
            from wxauto4.utils.win32 import SetClipboardText
            SetClipboardText(args.msg)
            
            time.sleep(1.0) # Ensure window transitions are complete
            log("Accessing WeChat internal edit box...")
            chat_api = wx._api._chat_api
            chat_api._activate_editbox()
            
            log("Injecting message natively...")
            chat_api.editbox.SendKeys('{Ctrl}v', waitTime=0.5)
            
            log("Striking pure RETURN natively into the focused editbox...")
            chat_api.editbox.SendKeys('{Enter}', waitTime=0.5)
            
            # Wait to confirm the text disappeared
            time.sleep(0.5)

            print(json.dumps({"status": "success", "detail": f"Sent to '{args.who}'"}))
        except Exception as e:
            print(json.dumps({"status": "error", "message": str(e)}))

if __name__ == "__main__":
    main()

import http.server
import socketserver
import json
import subprocess
import os

PORT = 18790

class PortalHandler(http.server.SimpleHTTPRequestHandler):
    def end_headers(self):
        # Enable CORS for local testing
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        super().end_headers()

    def do_OPTIONS(self):
        self.send_response(200)
        self.end_headers()

    def do_POST(self):
        if self.path == '/api/exec':
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            data = json.loads(post_data.decode('utf-8'))
            
            cmd = data.get('cmd')
            if not cmd:
                self.send_response(400)
                self.end_headers()
                self.wfile.write(b'{"status": "error", "message": "No command provided"}')
                return

            try:
                # Skill folder is usually one level up or at specific location
                # We aim for the workspace skills folder
                skills_dir = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", "skills", "wechat"))
                
                # Run the command and capture output
                result = subprocess.run(
                    cmd, 
                    shell=True,
                    capture_output=True, 
                    text=True,
                    cwd=skills_dir
                )
                
                output_str = result.stdout
                if result.stderr:
                    output_str += "\n" + result.stderr

                # Try to parse script output if it's JSON
                try:
                    response_data = json.loads(result.stdout)
                except ValueError:
                    response_data = {"status": "success", "output": output_str}
                
                self.send_response(200)
                self.send_header('Content-type', 'application/json')
                self.end_headers()
                self.wfile.write(json.dumps(response_data).encode('utf-8'))
            except Exception as e:
                self.send_response(500)
                self.send_header('Content-type', 'application/json')
                self.end_headers()
                self.wfile.write(json.dumps({"status": "error", "message": str(e)}).encode('utf-8'))
        else:
            self.send_response(404)
            self.end_headers()

if __name__ == "__main__":
    # Ensure we are in the portal directory to serve its files
    os.chdir(os.path.dirname(os.path.abspath(__file__)))
    with socketserver.TCPServer(("", PORT), PortalHandler) as httpd:
        print(f"ClawPortal Server running at http://127.0.0.1:{PORT}")
        httpd.serve_forever()
